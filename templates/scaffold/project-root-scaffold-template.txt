#!/usr/bin/env bash

set -euo pipefail

# ------------------------------------------------------------
# Purpose: Pre-commit is custom way to run various checks and scripts before committing changes to the repository.
#          This script is executed automatically by Git before a commit is finalized.
# Logic:
# 1. Set global Variables
# 2. Set logging functions
# 3. Ensure uv (Astral) is installed and available in PATH
# 4. Convert ODS to CSV and commit/push the generated CSVs to the repo.
# 5. Main function to orchestrate the execution of the pre-commit hook.
# ------------------------------------------------------------ 

# ------------------------------------------------------------
# 1. Global Variables
# ------------------------------------------------------------

SCRIPT_NAME="$(basename "$0")"
TS="$(date +%Y-%m-%dT%H:%M:%S%z)"

# Extend PATH to include ~/.local/bin for uv (Astral) in "$HOME/.local/bin" if it exists
export PATH="$HOME/.local/bin:$PATH"

# ------------------------------------------------------------
# 2. Set logging functions
# ------------------------------------------------------------

log_info() {
    echo "[INFO][$TS][$SCRIPT_NAME] $1"
}

log_error() {
    echo "[ERROR][$TS][$SCRIPT_NAME] $1" 
}

log_warning() {
    echo "[WARNING][$TS][$SCRIPT_NAME] $1"
}

log_debug() {
    echo "[DEBUG][$TS][$SCRIPT_NAME] $1"
}

# ------------------------------------------------------------
# 3. UV Astral check
# ------------------------------------------------------------
 
uv_astral_check() {

    # Store the path to uv (Astral) binary in UV_BIN variable
    UV_BIN="$(command -v uv 2>/dev/null || true)"

    # If uv is not found in PATH, check for a common installation path in $HOME/.local/bin
    if [[ -z "${UV_BIN}" && -x "${HOME}/.local/bin/uv" ]]; then
        # Store common uv installation path in UV_BIN for later use
        UV_BIN="${HOME}/.local/bin/uv"
    fi

    # If uv is still not found, check for a user-specific installation
    if [[ -z "${UV_BIN}" && -x "/home/${USER}/.local/bin/uv" ]]; then
        # Store user-specific uv installation path in UV_BIN for later use
        UV_BIN="/home/${USER}/.local/bin/uv"
    fi

    # If not found raise an error and provide instructions.
    if [[ -z "${UV_BIN}" ]]; then
        log_error "uv (Astral) is required for this pre-commit hook to run. Please install it and try again."
        log_info "uv is not installed or not in PATH."
        log_info "Install uv (Astral) and ensure it is available in shell PATH."
        log_info "  curl -LsSf https://astral.sh/uv/install.sh | sh"
        log_info "  source ~/.local/bin/env to update your PATH"
        log_info "HOME: ${HOME:-<unset>}"
        log_info "Current PATH: $PATH"
        exit 1
    fi    
}
 
# -----------------------------------------------------------------
# 4. Convert ODS to CSV and commit/push the generated CSVs to the repo.
# -----------------------------------------------------------------

convert_ods_to_csv() {

    # Check if the export_ods_to_csv.py script exists in the expected location
    if [[ -f "scripts/ci/export_ods_to_csv.py" ]]; then
        log_info "Found export_ods_to_csv.py script. Proceeding with ODS to CSV conversion."
    else
        log_error "export_ods_to_csv.py script not found in scripts/ci/. Please ensure the script exists."
        exit 1
    fi

    # Run the ODS to CSV conversion script with uv (Astral)
    "${UV_BIN}" run scripts/ci/export_ods_to_csv.py
 
    # Stage the generated CSV files for commit
    log_info "Staging generated CSV files for commit..."
    if find . -path '*/exportsCSV/*.csv' -type f 2>/dev/null | grep -q .; then
        git add -A -- ':(glob)**/exportsCSV/*.csv'
    else
        log_warning "No CSV files found in exportsCSV directories. Skipping git add."
    fi
}


# ----------------------------------------------------
# X. Main
# ----------------------------------------------------

main() {

    log_info "Starting pre-commit hook execution..."

    log_info "Checking for uv (Astral) installation..."
    uv_astral_check

    log_info "Running ODS to CSV conversion..."
    convert_ods_to_csv

    log_info "Pre-commit hook execution completed successfully."
}

main "$@"

#!/usr/bin/env bash

set -euo pipefail

# ================================================================
# Purpose: Enforce commit message convention
# Logic:
# 1. Set Global Variables
# 2. Logging functions
# 3. Check if commit message matches the pattern
# ================================================================

# ===============================================================
# 1. Variables
# ===============================================================
SCRIPT_NAME=$(basename "$0")
TS=$(date +%Y-%m-%dT%H:%M:%S%z)

commit_message=$(cat "$1")
pattern="^(feat|fix|chore|docs|test|refactor|ci|build)(\(.+\))?: .+"

PATH_TO_CONVENTIONS_DOCS="docs/03_project-governance/conventions/git-commit-conventions.md"

# ===============================================================
# 2. Logging functions
# ===============================================================

log_info(){
    echo "[INFO][$TS][$SCRIPT_NAME] $1"
}

log_error(){
    echo "[ERROR][$TS][$SCRIPT_NAME] $1"
}

log_warning(){
    echo "[WARNING][$TS][$SCRIPT_NAME] $1"
}

log_debug(){
    echo "[DEBUG][$TS][$SCRIPT_NAME] $1"
}


# ===============================================================
# 3. Check if commit message matches the pattern
# ===============================================================

if ! [[ $commit_message =~ $pattern ]]; then
    log_error "Invalid commit message:"
    log_error "$commit_message"
    echo ""
    log_error "Expected format:"
    log_error "type: description"
    log_error "Check the commit message conventions in $PATH_TO_CONVENTIONS_DOCS"
    exit 1
fi

# ------------------------------------------------------------------------------
# Purpose: This workflow is responsible for exporting ODS files to CSV format and 
# verifying that the generated CSV files are up to date with the committed versions. 
# ------------------------------------------------------------------------------ 
name: Export ODS to CSV

# This workflow is triggered when called by another workflow using the "workflow_call" event.
on:
  workflow_call:

jobs:
    export:
        runs-on: ubuntu-latest
        # Set permissions to read the repository contents, which is necessary for checking out the code and verifying CSV files.
        permissions:
            contents: read
        steps:
            - uses: actions/checkout@v4
              # Set lfs to true to ensure that any large files managed by Git LFS are properly checked out.
              with:
                lfs: true

              # Install Python dependencies required for the export process, including pandas for data manipulation and odfpy for handling ODS files.  
            - name: Install Python dependencies
              run: |
                python -m pip install --upgrade pip
                python -m pip install pandas odfpy

              # The job step to export ODS files to CSV format using a Python script. 
              # This script reads ODS files and generates corresponding CSV files in the specified output directory.
            - name: Verify CSVs are up to date
              shell: bash
              run: |
                set -euo pipefail

                # Re-generate CSVs from ODS files in the CI working directory.
                python scripts/ci/export_ods_to_csv.py

                # Fail if any CSV differs from what is committed.
                if [[ -n "$(git status --porcelain)" ]]; then
                  echo ""
                  echo "ERROR: The following CSV files are out of date:"
                  git status --porcelain
                  echo ""
                  echo "Run the pre-commit hook locally and commit the updated CSVs:"
                  echo "  git config core.hooksPath .githooks"
                  exit 1
                fi

                echo "All CSV files are up to date."

# Orchestration workflow for all CI jobs. 
# This workflow is triggered on push, pull request and manual dispatch.
name: ci-orchestration

on:
  push:
  pull_request:
  workflow_dispatch:

jobs:
  # The "changes" job detects if any relevant files have changed in the current commit or pull request.
  # If so, it sets an output variable that can be used by downstream jobs to determine whether they should run.
  changes:
    runs-on: ubuntu-latest
    outputs:
      # The output variable "ods_pipeline" is set to true if any relevant files have changed, and false otherwise.
      ods_pipeline: ${{ steps.filter.outputs.ods_pipeline }}
    steps:
      - uses: actions/checkout@v4

      - name: Detect relevant changes
        id: filter
        uses: dorny/paths-filter@v3
        with:
          # The "ods_pipeline" filter checks for changes in the following files and directories:
          filters: |
            ods_pipeline:
              - "**/*.ods"
              - "**/exportsCSV/*.csv"
              - ".githooks/pre-commit"
              - "scripts/ci/export_ods_to_csv.py"
              - ".github/workflows/ci-export_ods_to_csv.yml"
              - ".github/workflows/ci.yml"

  # The "export-ods-to-csv" job runs the export process if relevant files have changed or 
  # if the workflow was manually triggered. For job details, see the ci-export-ods-to-csv.yml workflow file.
  # Policy: CSV files under exportsCSV are generated artifacts and must match ODS + exporter output.             
  export-ods-to-csv:
     needs: [changes]
     if: github.event_name == 'workflow_dispatch' || needs.changes.outputs.ods_pipeline == 'true'
     uses: ./.github/workflows/ci-export_ods_to_csv.yml

  # Placeholder job for linting. This job runs after the export-ods-to-csv job and is defined in the ci-lint.yml workflow file.   
  lint:
     needs: export-ods-to-csv
     uses: ./.github/workflows/ci-lint.yml

  # Placeholder job for testing. This job runs after the lint job and is defined in the ci-test.yml workflow file.   
  test:
     needs: lint
     uses: ./.github/workflows/ci-test.yml

  # Placeholder job for building. This job runs after the test job and is defined in the ci-build.yml workflow file.   
  build:
     needs: test
     uses: ./.github/workflows/ci-build.yml
